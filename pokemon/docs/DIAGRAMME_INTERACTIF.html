initGame()

Rôle : Point d'entrée principal qui orchestre le démarrage du jeu

Appelle les fonctions :

  • → loadRandomPokemon() (×2 fois)
  • → initializePokemon() (×2 fois)
  • → displayPokemon() (×2 fois)
  • → updateTurnIndicator()
  • → addLog()
async function initGame() { // Charger 2 Pokémon const [poke1, poke2] = await Promise.all([ loadRandomPokemon(), loadRandomPokemon() ]); // Initialiser gameState.pokemon1 = initializePokemon(poke1); gameState.pokemon2 = initializePokemon(poke2); // Afficher displayPokemon(1, gameState.pokemon1); displayPokemon(2, gameState.pokemon2); updateTurnIndicator(); addLog("Le combat commence !"); }

loadRandomPokemon()

Rôle : Charge un Pokémon aléatoire depuis l'API PokéAPI

Retourne : Objet avec les données brutes de l'API

Appelle : Aucune fonction (fonction feuille)

async function loadRandomPokemon() { const randomId = Math.floor(Math.random() * 151) + 1; const response = await fetch( `https://pokeapi.co/api/v2/pokemon/${randomId}` ); const data = await response.json(); return data; }

initializePokemon(apiData)

Rôle : Transforme les données API en objet Pokémon utilisable

Paramètres : apiData (données brutes de l'API)

Retourne : Objet Pokémon avec stats de combat

Appelle : Aucune fonction (fonction feuille)

function initializePokemon(apiData) { const hp = apiData.stats.find(s => s.stat.name === 'hp').base_stat; const attack = apiData.stats.find(s => s.stat.name === 'attack').base_stat; return { name: apiData.name, sprite: apiData.sprites.front_default, maxHp: CONFIG.MAX_HP, currentHp: CONFIG.MAX_HP, currentMana: CONFIG.INITIAL_MANA, attack: attack, // ... etc }; }

displayPokemon(pokemonNumber, pokemon)

Rôle : Affiche toutes les informations d'un Pokémon dans sa carte

Paramètres : pokemonNumber (1 ou 2), pokemon (objet)

Appelle :

  • → updateBars()
function displayPokemon(pokemonNumber, pokemon) { document.getElementById(`pokemon${pokemonNumber}-name`) .textContent = pokemon.name; document.getElementById(`pokemon${pokemonNumber}-image`) .src = pokemon.sprite; // Afficher les stats document.getElementById(`pokemon${pokemonNumber}-attack`) .textContent = pokemon.attack; updateBars(pokemonNumber); }

updateBars(pokemonNumber)

Rôle : Met à jour les barres de vie et mana avec les valeurs actuelles

Paramètres : pokemonNumber (1 ou 2)

Appelle : Aucune fonction (fonction feuille)

function updateBars(pokemonNumber) { const pokemon = gameState[`pokemon${pokemonNumber}`]; // Calculer les pourcentages const hpPercent = (pokemon.currentHp / pokemon.maxHp) * 100; // Mettre à jour la barre const hpBar = document.getElementById(`pokemon${pokemonNumber}-hp-bar`); hpBar.style.width = `${hpPercent}%`; // Changer la couleur if (hpPercent > 50) hpBar.classList.add('high'); else if (hpPercent > 25) hpBar.classList.add('medium'); else hpBar.classList.add('low'); }

updateTurnIndicator()

Rôle : Met à jour l'affichage pour montrer quel Pokémon joue

Appelle : Aucune fonction (fonction feuille)

function updateTurnIndicator() { const pokemon = gameState[`pokemon${gameState.currentTurn}`]; document.getElementById('turn-indicator').textContent = `Tour de ${pokemon.name.toUpperCase()}`; // Bordure dorée sur la carte active document.getElementById('pokemon1-card') .classList.remove('active-turn'); document.getElementById('pokemon2-card') .classList.remove('active-turn'); document.getElementById(`pokemon${gameState.currentTurn}-card`) .classList.add('active-turn'); }

addLog(message, type)

Rôle : Ajoute un message au log de combat

Paramètres : message (texte), type ('attack', 'heal', 'victory')

Appelle : Aucune fonction (fonction feuille)

function addLog(message, type = 'attack') { const logContainer = document.getElementById('log-container'); const logEntry = document.createElement('div'); logEntry.className = `log-entry ${type}`; logEntry.textContent = message; logContainer.insertBefore(logEntry, logContainer.firstChild); // Limiter à 10 messages while (logContainer.children.length > 10) { logContainer.removeChild(logContainer.lastChild); } }

basicAttack()

Rôle : Exécute une attaque simple (dégâts + recharge mana)

Appelle :

  • → animateAttack()
  • → animateDamage()
  • → updateBars() (×2 fois)
  • → addLog()
  • → checkVictory()
  • → nextTurn()
function basicAttack() { const attacker = gameState[`pokemon${gameState.currentTurn}`]; const defenderNum = gameState.currentTurn === 1 ? 2 : 1; const defender = gameState[`pokemon${defenderNum}`]; animateAttack(gameState.currentTurn); // Appliquer dégâts defender.currentHp -= CONFIG.BASIC_ATTACK_DAMAGE; // Recharger mana attacker.currentMana += CONFIG.BASIC_MANA_GAIN; animateDamage(defenderNum); updateBars(gameState.currentTurn); updateBars(defenderNum); addLog(`${attacker.name} attaque !`, 'attack'); if (checkVictory()) return; nextTurn(); }

magicAttack()

Rôle : Exécute une attaque magique (gros dégâts - coût mana)

Appelle :

  • → animateAttack()
  • → animateDamage()
  • → updateBars() (×2 fois)
  • → addLog()
  • → checkVictory()
  • → nextTurn()
function magicAttack() { const attacker = gameState[`pokemon${gameState.currentTurn}`]; // Vérifier le mana if (attacker.currentMana < CONFIG.MAGIC_MANA_COST) { addLog("Pas assez de mana !", 'attack'); return; } const defenderNum = gameState.currentTurn === 1 ? 2 : 1; const defender = gameState[`pokemon${defenderNum}`]; animateAttack(gameState.currentTurn); // Consommer mana et infliger dégâts attacker.currentMana -= CONFIG.MAGIC_MANA_COST; defender.currentHp -= CONFIG.MAGIC_ATTACK_DAMAGE; animateDamage(defenderNum); updateBars(gameState.currentTurn); updateBars(defenderNum); addLog(`${attacker.name} attaque magique !`, 'attack'); if (checkVictory()) return; nextTurn(); }

nextTurn()

Rôle : Passe au tour du Pokémon suivant

Appelle :

  • → updateTurnIndicator()
function nextTurn() { gameState.currentTurn = gameState.currentTurn === 1 ? 2 : 1; updateTurnIndicator(); }

checkVictory()

Rôle : Vérifie si un Pokémon a gagné

Retourne : Boolean (true si partie terminée)

Appelle :

  • → endGame()
function checkVictory() { if (gameState.pokemon1.currentHp <= 0) { endGame(2); return true; } if (gameState.pokemon2.currentHp <= 0) { endGame(1); return true; } return false; }

endGame(winner)

Rôle : Termine la partie et affiche le vainqueur

Paramètres : winner (numéro du Pokémon gagnant 1 ou 2)

Appelle :

  • → addLog()
function endGame(winner) { gameState.gameOver = true; const winnerPokemon = gameState[`pokemon${winner}`]; // Désactiver les boutons document.getElementById('btn-basic-attack').disabled = true; document.getElementById('btn-magic-attack').disabled = true; addLog(`${winnerPokemon.name} GAGNE !`, 'victory'); // Afficher la modal de victoire document.getElementById('victory-modal').classList.add('active'); }

🔄 Flux d'exécution lors d'une attaque
1. Clic bouton
basicAttack() ou magicAttack()
2. Animation attaquant
animateAttack(attacker)
3. Calcul des dégâts
Modification HP/Mana dans gameState
4. Animation défenseur
animateDamage(defender)
5. Mise à jour affichage
updateBars(attacker)
updateBars(defender)
6. Log de combat
addLog(message)
7. Vérification victoire
checkVictory() → endGame() si victoire
8. Tour suivant
nextTurn() → updateTurnIndicator()
